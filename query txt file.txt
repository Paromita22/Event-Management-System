CREATE TABLE Customer (
    cust_id varchar(20)PRIMARY KEY ,
    cust_name VARCHAR(100),
    cust_phn VARCHAR(15),
    cust_email VARCHAR(100) UNIQUE
);


CREATE TABLE Eventt (
    evnt_id varchar(20) PRIMARY KEY ,
    evnt_type VARCHAR(100),
    evnt_budget numeric(10, 2),
    evnt_date DATE
);


CREATE TABLE Organizer (
    org_id varchar(20) PRIMARY KEY ,
    org_name VARCHAR(100),
    org_mail VARCHAR(100) UNIQUE,
    org_phn VARCHAR(15),
    org_provided_services varchar(500)
);


CREATE TABLE Venue (
    ven_id varchar(20) PRIMARY KEY,
    ven_address VARCHAR(255),
    ven_capacity numeric
);

CREATE TABLE Feedback (
    fed_id varchar(20) PRIMARY KEY ,
    rating INT CHECK (rating BETWEEN 1 AND 5),
    review varchar(500)
   
);

CREATE TABLE Booking (
    booking_id varchar(20) PRIMARY KEY ,
    cust_id varchar(20),
    evnt_id varchar(20),
    booking_date DATE,
    FOREIGN KEY (cust_id) REFERENCES Customer(cust_id),
    FOREIGN KEY (evnt_id) REFERENCES Eventt(evnt_id)
);


CREATE TABLE Organises (
    org_event_id varchar(20) PRIMARY KEY ,
    org_id varchar(20),
    evnt_id varchar(20),
    FOREIGN KEY (org_id) REFERENCES Organizer(org_id),
    FOREIGN KEY (evnt_id) REFERENCES Eventt(evnt_id)
);

CREATE TABLE Availability (
    avail_id varchar(20) PRIMARY KEY,
    evnt_id varchar(20),
    ven_id varchar(20),
    ven_status VARCHAR(50),
    avail_date DATE,
    start_time TIME,
    end_time TIME,
    FOREIGN KEY (evnt_id) REFERENCES Eventt(evnt_id),
    FOREIGN KEY (ven_id) REFERENCES Venue(ven_id)
);

CREATE TABLE OrganizerFeedback (
    org_feedback_id varchar(20) PRIMARY KEY ,
    org_id varchar(20),
    cust_id varchar(20),
    fed_id varchar(20),
    FOREIGN KEY (org_id) REFERENCES Organizer(org_id),
    FOREIGN KEY (cust_id) REFERENCES Customer(cust_id),
    FOREIGN KEY (fed_id) REFERENCES Feedback(fed_id)
);

CREATE TABLE CustomerFeedback (
    cust_feedback_id varchar(20) PRIMARY KEY ,
    cust_id varchar(20),
    org_id varchar(20),
    fed_id varchar(20),
    FOREIGN KEY (cust_id) REFERENCES Customer(cust_id),
    FOREIGN KEY (org_id) REFERENCES Organizer(org_id),
    FOREIGN KEY (fed_id) REFERENCES Feedback(fed_id)
);
